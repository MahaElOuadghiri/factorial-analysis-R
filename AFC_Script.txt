```r
# Create the contingency table
contingency_table <- matrix(
  c(45, 25, 30,
    20, 45, 35,
    30, 30, 40,
    15, 60, 25),
  nrow = 4,
  byrow = TRUE
)

# Add row and column names
rownames(contingency_table) <- c(
  "Sciences", "Arts and Humanities", "Economics", "Engineering"
)

colnames(contingency_table) <- c(
  "In-person", "Distance Learning", "Hybrid"
)

# Display the contingency table
print(contingency_table)


# Load the libraries
library(FactoMineR)
library(factoextra)

# Perform Correspondence Analysis (CA)
res <- CA(contingency_table, graph = FALSE)

# Display eigenvalues and inertia
print(res$eig)

# Plot the factorial map
fviz_ca_biplot(res, repel = TRUE)
```
